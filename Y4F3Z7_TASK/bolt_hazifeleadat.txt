1. Feladat – XML adatok kiolvasása XMLTABLE segítségével
----------------------------------------------------------------

-- a) Az összes ügyfél kilistázása
SELECT x.cid,
       x.nev,
       x.email,
       x.telefonszam,
       x.varos,
       x.utca,
       x.hazszam
FROM bolt_xml b,
     XMLTABLE('/bolt/ugyfel'
         PASSING b.adat
         COLUMNS
             cid         VARCHAR2(10)  PATH '@cid',
             nev         VARCHAR2(200) PATH 'nev',
             email       VARCHAR2(200) PATH 'email',
             telefonszam VARCHAR2(20)  PATH 'telefonszam',
             varos       VARCHAR2(100) PATH 'cim/varos',
             utca        VARCHAR2(200) PATH 'cim/utca',
             hazszam     VARCHAR2(10)  PATH 'cim/hazszam'
     ) x;

-- b) Szűrés: csak a kifizetett rendelések
SELECT x.rid,
       x.ugyfel_ref,
       x.rendeles_datum,
       x.osszeg
FROM bolt_xml b,
     XMLTABLE('/bolt/rendeles'
         PASSING b.adat
         COLUMNS
             rid            VARCHAR2(10)  PATH '@rid',
             ugyfel_ref     VARCHAR2(10)  PATH '@ugyfel_ref',
             rendeles_datum VARCHAR2(20)  PATH 'rendeles_datum',
             fizetve        VARCHAR2(10)  PATH 'fizetve',
             osszeg         NUMBER        PATH 'osszeg'
     ) x
WHERE x.fizetve = 'true';

-- c) Rendelések kilistázása az ügyfél nevével együtt (JOIN-szerű XMLTABLE)
SELECT r.rid,
       r.rendeles_datum,
       r.osszeg,
       u.nev AS ugyfel_nev
FROM bolt_xml b,
     XMLTABLE('/bolt/rendeles'
         PASSING b.adat
         COLUMNS
             rid            VARCHAR2(10) PATH '@rid',
             ugyfel_ref     VARCHAR2(10) PATH '@ugyfel_ref',
             rendeles_datum VARCHAR2(20) PATH 'rendeles_datum',
             osszeg         NUMBER       PATH 'osszeg'
     ) r,
     XMLTABLE('/bolt/ugyfel'
         PASSING b.adat
         COLUMNS
             cid VARCHAR2(10)  PATH '@cid',
             nev VARCHAR2(200) PATH 'nev'
     ) u
WHERE r.ugyfel_ref = u.cid;

2. Feladat – UDT (Object Type) létrehozása az XML struktúra alapján
----------------------------------------------------------------

-- a) Cím típus (az XML <cim> elem alapján)
CREATE OR REPLACE TYPE bolt_cim_t AS OBJECT (
    orszag       VARCHAR2(100),
    varos        VARCHAR2(100),
    iranyitoszam VARCHAR2(10),
    utca         VARCHAR2(200),
    hazszam      VARCHAR2(10)
);
/

-- b) Ügyfél típus beágyazott cím objektummal és tagfüggvénnyel
CREATE OR REPLACE TYPE ugyfel_t AS OBJECT (
    cid         VARCHAR2(10),
    nev         VARCHAR2(200),
    email       VARCHAR2(200),
    telefonszam VARCHAR2(20),
    ceg         VARCHAR2(5),
    cim         bolt_cim_t,
    MEMBER FUNCTION leiras RETURN VARCHAR2
);
/

-- c) TYPE BODY – tagfüggvény implementálása
CREATE OR REPLACE TYPE BODY ugyfel_t AS
    MEMBER FUNCTION leiras RETURN VARCHAR2 IS
    BEGIN
        RETURN SELF.nev || ' (' || SELF.email || ') – ' || SELF.cim.varos;
    END leiras;
END;
/


----------------------------------------------------------------
3. Feladat – Object Table létrehozása és feltöltése XML adatokból
----------------------------------------------------------------

-- a) Object Table létrehozása
CREATE TABLE ugyfelek OF ugyfel_t (
    PRIMARY KEY (cid)
);

-- b) Feltöltés XML-ből: INSERT … SELECT + XMLTABLE
INSERT INTO ugyfelek
SELECT ugyfel_t(
    x.cid,
    x.nev,
    x.email,
    x.telefonszam,
    x.ceg,
    bolt_cim_t(x.orszag, x.varos, x.iranyitoszam, x.utca, x.hazszam)
)
FROM bolt_xml b,
     XMLTABLE('/bolt/ugyfel'
         PASSING b.adat
         COLUMNS
             cid          VARCHAR2(10)  PATH '@cid',
             nev          VARCHAR2(200) PATH 'nev',
             email        VARCHAR2(200) PATH 'email',
             telefonszam  VARCHAR2(20)  PATH 'telefonszam',
             ceg          VARCHAR2(5)   PATH 'ceg',
             orszag       VARCHAR2(100) PATH 'cim/orszag',
             varos        VARCHAR2(100) PATH 'cim/varos',
             iranyitoszam VARCHAR2(10)  PATH 'cim/iranyitoszam',
             utca         VARCHAR2(200) PATH 'cim/utca',
             hazszam      VARCHAR2(10)  PATH 'cim/hazszam'
     ) x;
COMMIT;

-- c) Ellenőrzés: az Object Table tartalma
SELECT u.cid, u.nev, u.email,
       u.cim.varos, u.cim.utca, u.cim.hazszam
FROM ugyfelek u;


----------------------------------------------------------------
4. Feladat – Lekérdezések az Object Table-ön
----------------------------------------------------------------

-- a) Tagfüggvény hívása lekérdezésben
SELECT u.leiras() AS leiras
FROM ugyfelek u;

-- b) Aggregáció – ügyfelek száma városonként
SELECT u.cim.varos AS varos,
       COUNT(*)    AS ugyfelek_szama
FROM ugyfelek u
GROUP BY u.cim.varos
ORDER BY ugyfelek_szama DESC;

6. Feladat – Kollekciós típus: az áru értékelései
----------------------------------------------------------------
-- Az XML-ben az <aru> elemnek több <ertek> gyermeke is lehet.
-- Ezt Oracle VARRAY típussal modellezzük.

-- a) VARRAY típus létrehozása (max 10 értékelés)
CREATE OR REPLACE TYPE ertekelesek_va AS VARRAY(10) OF NUMBER;
/

-- b) Áru típus VARRAY-jel
CREATE OR REPLACE TYPE aru_t AS OBJECT (
    aid            VARCHAR2(10),
    nev            VARCHAR2(200),
    gyarto         VARCHAR2(200),
    kategoria      VARCHAR2(100),
    raktarkeszlet  NUMBER,
    garancia_honap NUMBER,
    ertekelesek    ertekelesek_va
);
/

CREATE TABLE aruk OF aru_t (
    PRIMARY KEY (aid)
);

-- c) Feltöltés kézzel (az XML ertekelesek adatai alapján)
INSERT INTO aruk VALUES (
    aru_t('a10', 'Laptop', 'Lenovo', 'Elektronika', 5, 24,
        ertekelesek_va(5, 5, 4))
);
COMMIT;

-- d) VARRAY elemeinek lekérdezése TABLE() operátorral
SELECT a.nev, a.gyarto, v.COLUMN_VALUE AS ertek
FROM aruk a,
     TABLE(a.ertekelesek) v;

-- e) Átlagos értékelés árunként (TABLE() + AVG)
SELECT a.nev,
       AVG(v.COLUMN_VALUE) AS atlag_ertekeles
FROM aruk a,
     TABLE(a.ertekelesek) v
GROUP BY a.aid, a.nev
ORDER BY atlag_ertekeles DESC;
